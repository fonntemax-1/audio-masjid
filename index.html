let keuanganAnimId = null; let keuanganPos = 0; function startKeuanganScroll() { const area = document.getElementById( 'keuangan-scroll-area' ); const content = document.getElementById( 'keuangan-content' ); if (!area || !content) return; if (keuanganAnimId) { cancelAnimationFrame(keuanganAnimId); keuanganAnimId = null; } keuanganPos = 0; content.style.transform = 'translateY(0px)'; const finPanelBox = document.getElementById('fin-panel-box'); const isQurbanScroll = !!finPanelBox && finPanelBox.classList.contains('qurban-display'); // Jarak awal area scrolling harus benar-benar terlihat di bawah // TANGGAL. Margin CSS pada #fin-sub tidak cukup karena tabel // menggunakan position:absolute dan transform:translateY(). // Target KEUANGAN: TOP tabel = 1x tinggi/font TANGGAL. // QUR'BAN mempertahankan pattern yang sudah ada. let keuanganStartGapPx = 0; const dateElForScrollGap = isQurbanScroll ? document.getElementById('fin-extra') : document.getElementById('fin-sub'); if (dateElForScrollGap) { const dateFontPx = parseFloat( window.getComputedStyle(dateElForScrollGap).fontSize ) || 16; keuanganStartGapPx = Math.max( 1, Math.round(dateFontPx) ); } // Posisi awal tabel benar-benar turun 1 font TANGGAL. // Ini berlaku untuk KEUANGAN dan mempertahankan QUR'BAN. content.style.top = '0px'; content.style.paddingTop = '0px'; content.style.boxSizing = 'border-box'; content.style.transform = 'translateY(' + keuanganStartGapPx + 'px)'; /* * Loop kontinu dengan jeda kosong 1/2 tinggi frame. * Setelah satu tabel selesai keluar, layar kosong terlebih * dahulu setengah frame. Baru salinan tabel berikutnya masuk. * Ini membuat pergantian lebih nyaman dibaca tanpa refresh. */ let firstBlock = content.firstElementChild; if ( firstBlock && firstBlock.tagName === 'TABLE' && content.children.length === 1 ) { const clone = firstBlock.cloneNode(true); clone.setAttribute('aria-hidden', 'true'); const spacer = document.createElement('div'); spacer.className = 'keuangan-scroll-spacer'; let spacerHeight = Math.max( 1, Math.round(area.clientHeight * 0.5) ); // QUR'BAN: beri ruang END scrolling lebih longgar. // Tambahkan 4 ukuran font tanggal agar baris terakhir // tidak terlalu mepet dengan batas atas saat siklus selesai. // Jeda standar laporan keuangan tetap dipertahankan. if (isQurbanScroll) { // QUR'BAN: END scrolling sekitar 1/2 frame, // agar ada jeda yang nyaman sebelum blok pertama muncul lagi. spacerHeight = Math.max( 1, Math.round(area.clientHeight * 0.5) ); } spacer.style.height = spacerHeight + 'px'; spacer.setAttribute('aria-hidden', 'true'); content.appendChild(spacer); content.appendChild(clone); } firstBlock = content.firstElementChild; if (!firstBlock || firstBlock.tagName !== 'TABLE') return; const loopSpacer = content.children[1]; const loopHeight = firstBlock.getBoundingClientRect().height; const gapHeight = loopSpacer && loopSpacer.classList.contains('keuangan-scroll-spacer') ? loopSpacer.getBoundingClientRect().height : Math.round(area.clientHeight * 0.5); if (loopHeight <= area.clientHeight) { content.style.transform = 'translateY(0px)'; return; } const loopDistance = loopHeight + gapHeight; const speed = 0.2; let lastFrame = 0; function step(timestamp) { if ( lastFrame && (timestamp - lastFrame) < (1000 / 30) ) { keuanganAnimId = requestAnimationFrame(step); return; } lastFrame = timestamp; keuanganPos -= speed; /* * Reset hanya setelah tabel pertama + jeda 1/2 frame * sudah seluruhnya lewat. Pada saat reset, salinan kedua * berada tepat di posisi awal tabel pertama. */ if (Math.abs(keuanganPos) >= loopDistance) { // Setiap siklus dimulai kembali dengan jarak 1x font TANGGAL // sebelum TOP tabel. Ini berlaku juga untuk KEUANGAN biasa. keuanganPos = keuanganStartGapPx; } content.style.transform = 'translateY(' + keuanganPos + 'px)'; keuanganAnimId = requestAnimationFrame(step); } keuanganAnimId = requestAnimationFrame(step); }
